// 安全的程式碼（已修復 XSS） function escapeHtml(text) { var map = { '&': '&', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' }; return text.replace(/[&<>"']/g, function(m) { return map[m]; }); } function search() { var keyword = document.getElementById('searchInput').value; var result = document.getElementById('resultContent'); // ✅ 安全！先進行 HTML 編碼再顯示 var safeKeyword = escapeHtml(keyword); result.innerHTML = '你搜尋了：<strong>' + safeKeyword + '</strong>'; }

🛡️ 防禦措施：

  • HTML 編碼：將特殊字元轉換為 HTML 實體
  • 輸入驗證：檢查並過濾不合法的輸入
  • 使用 textContent：對於純文字可使用 textContent 代替 innerHTML
  • CSP 政策：設定 Content-Security-Policy 限制腳本執行
